"""Local, machine-specific integration pilot. No credentials or raw model reasoning are saved."""
from __future__ import annotations
import json
import os
from pathlib import Path
import queue
import secrets
import shutil
import subprocess
import sys
import threading
import time
import tomllib
from http.server import BaseHTTPRequestHandler, HTTPServer

from pomdp_bench.collection import resume_suite, read_run
from pomdp_bench.storage import write_json

ROOT = Path(__file__).resolve().parent
EMPTY = ROOT / 'reserve-pilot-empty-workspace'
EMPTY.mkdir(exist_ok=True)
MODELS = {'gpt-5.6-sol': 'openai', 'custom/z-ai/glm-5.3': 'codex-router'}


class Client:
    def __init__(self):
        config_path = Path(os.environ.get('CODEX_HOME', Path.home()/'.codex')) / 'config.toml'
        config = tomllib.loads(config_path.read_text(encoding='utf-8'))
        command = [shutil.which('codex'), 'app-server', '--stdio']
        flags = ['plugins','apps','multi_agent','multi_agent_v2','shell_tool','unified_exec','computer_use',
                 'browser_use','browser_use_external','in_app_browser','view_image','code_mode','code_mode_host',
                 'goals','sleep_tool','hooks','workspace_dependencies','skill_search','memories',
                 'unbounded_connection_retries','tool_suggest']
        for flag in flags:
            command += ['--disable', flag]
        settings = {'web_search': 'disabled', 'tools.view_image': False, 'project_doc_max_bytes': 0,
                    'model_reasoning_effort': 'high', 'model_reasoning_summary': 'none',
                    'model_providers.codex-router.request_max_retries': 0,
                    'model_providers.codex-router.stream_max_retries': 0}
        for name in config.get('mcp_servers', {}):
            settings[f'mcp_servers.{name}.enabled'] = False
        for name, value in settings.items():
            command += ['-c', name+'='+json.dumps(value)]
        self.process = subprocess.Popen(command, stdin=subprocess.PIPE, stdout=subprocess.PIPE,
                                        stderr=subprocess.DEVNULL, text=True, encoding='utf-8',
                                        cwd=EMPTY, creationflags=subprocess.CREATE_NO_WINDOW if os.name=='nt' else 0)
        self.pending, self.condition, self.events, self.counter = {}, threading.Condition(), queue.Queue(), 0
        self.lock = threading.Lock()
        self.reader = threading.Thread(target=self.read, daemon=True)
        self.reader.start()
        try:
            self.rpc('initialize', {'clientInfo': {'name':'public-json-pilot','version':'1.0'},
                                    'capabilities':{'experimentalApi':True}})
        except BaseException:
            self.close()
            raise
        self.send({'method':'initialized','params':{}})
        self.observed = []

    def send(self, value):
        self.process.stdin.write(json.dumps(value)+'\n')
        self.process.stdin.flush()

    def read(self):
        for line in self.process.stdout:
            try: message=json.loads(line)
            except ValueError: continue
            if 'id' in message and 'method' in message:
                self.send({'id':message['id'], 'error':{'code':-32601,'message':'No client tools are available'}})
            elif 'id' in message:
                with self.condition:
                    self.pending[message['id']]=message
                    self.condition.notify_all()
            else:
                self.events.put(message)

    def rpc(self, method, params, timeout=20):
        self.counter += 1
        ident=self.counter
        self.send({'id':ident,'method':method,'params':params})
        with self.condition:
            if not self.condition.wait_for(lambda:ident in self.pending, timeout):
                raise RuntimeError('RPC timeout: '+method)
            response=self.pending.pop(ident)
        if 'error' in response:
            # Method and code only; never print a provider body or configuration.
            raise RuntimeError('RPC rejected '+method+' code '+str(response['error'].get('code')))
        return response['result']

    def act(self, body):
        with self.lock:
            model=body['model']
            if model not in MODELS or body.get('reasoning_effort')!='high':
                raise ValueError('Unexpected pilot configuration')
            result=self.rpc('thread/start', {'model':model,'modelProvider':MODELS[model],
                'allowProviderModelFallback':False,'ephemeral':True,'cwd':str(EMPTY),
                'environments':[],'runtimeWorkspaceRoots':[],'selectedCapabilityRoots':[],
                'dynamicTools':[],'approvalPolicy':'never',
                'baseInstructions':body['messages'][0]['content'],
                'developerInstructions':'Use only the supplied public JSON task and history. Return one JSON action.',
                'config':{'model_reasoning_effort':'high','project_doc_max_bytes':0}})
            if (result['model'] != model or result['modelProvider'] != MODELS[model]
                    or result.get('reasoningEffort') != 'high'):
                raise ValueError('Model, provider or reasoning configuration mismatch')
            sources=result.get('instructionSources',[])
            tid=result['thread']['id']
            meta={'model':result['model'],'provider':result['modelProvider'],
                  'reasoning_effort':result.get('reasoningEffort'),
                  'instruction_sources':sources,'environments':[]}
            self.observed.append(meta)
            turn=self.rpc('turn/start',{'threadId':tid,'environments':[],'effort':'high',
                'input':[{'type':'text','text':body['messages'][1]['content'],'text_elements':[]}]})
            turn_id=turn['turn']['id']
            deadline=time.monotonic()+150
            final, usage = None, None
            try:
                while True:
                    remaining=deadline-time.monotonic()
                    if remaining <= 0:
                        raise TimeoutError('Local bridge event deadline exceeded')
                    event=self.events.get(timeout=remaining)
                    params=event.get('params',{})
                    if params.get('threadId')!=tid:
                        continue
                    method=event.get('method')
                    if method=='thread/tokenUsage/updated':
                        u=params['tokenUsage']['total']
                        usage={'prompt_tokens':u['inputTokens'],'completion_tokens':u['outputTokens']}
                        # App-server defaults absent reasoning counts to zero: do not claim presence.
                    if method=='item/completed':
                        item=params['item']
                        if item['type']=='agentMessage':
                            final=item.get('text')
                        elif item['type'] not in ('userMessage','reasoning'):
                            raise RuntimeError('Unexpected tool or non-message item: '+item['type'])
                    if method=='turn/completed':
                        if params['turn']['status']!='completed':
                            raise RuntimeError('Model turn did not complete')
                        if final is None:
                            raise RuntimeError('No final model action')
                        action=json.loads(final)
                        if not isinstance(action,dict):
                            raise RuntimeError('Non-object action')
                        return {'choices':[{'message':{'content':final}}], 'usage':usage}
            finally:
                try:
                    self.rpc('turn/interrupt',{'threadId':tid,'turnId':turn_id},timeout=2)
                except Exception:
                    pass
                try:
                    self.rpc('thread/unsubscribe',{'threadId':tid},timeout=2)
                except Exception:
                    pass

    def close(self):
        self.process.terminate()
        try:self.process.wait(timeout=8)
        except subprocess.TimeoutExpired:self.process.kill()


def main():
    out=Path(sys.argv[1]).resolve()
    manifest, _ = read_run(out, partial=True)
    if manifest.get('study', {}).get('plan', {}).get('study_id') != 'verification-reserve-live-pilot-v1':
        raise ValueError('Use the prepared preregistered pilot only')
    client=Client()
    events=[]
    token=secrets.token_hex(20)
    class Handler(BaseHTTPRequestHandler):
        def log_message(self,*args):pass
        def do_POST(self):
            started=time.monotonic()
            if self.headers.get('Authorization')!='Bearer '+token:
                self.send_error(401)
                return
            body=json.loads(self.rfile.read(int(self.headers['Content-Length'])))
            event={'model':body.get('model'),'request':len(events)+1}
            try:
                value=client.act(body)
                event.update(status='ok',elapsed_seconds=round(time.monotonic()-started,3))
                code=200
            except Exception as exc:
                event.update(status='error',elapsed_seconds=round(time.monotonic()-started,3),error_type=type(exc).__name__,
                             diagnostic=str(exc) if isinstance(exc,RuntimeError) else 'Rejected local bridge response')
                value={'error':{'message':'Local bridge request failed'}}
                code=502
            events.append(event)
            write_json(ROOT/'reserve-pilot-transport-events.json',events)
            write_json(ROOT/'reserve-pilot-observed-config.json',client.observed)
            print(json.dumps(event),flush=True)
            raw=json.dumps(value).encode()
            try:
                self.send_response(code);self.send_header('Content-Type','application/json')
                self.send_header('Content-Length',str(len(raw)));self.end_headers();self.wfile.write(raw)
            except OSError:pass
    server=HTTPServer(('127.0.0.1',0),Handler)
    thread=threading.Thread(target=server.serve_forever,daemon=True)
    thread.start()
    os.environ['PILOT_ENDPOINT']=f'http://127.0.0.1:{server.server_port}/v1/chat/completions'
    os.environ['PILOT_KEY']=token
    try:
        report=resume_suite(out)
        print(json.dumps({'complete':report['complete'],'overall':report['overall'],'study_analysis':report['study_analysis']},ensure_ascii=False),flush=True)
    finally:
        server.shutdown();server.server_close();client.close()

if __name__=='__main__':main()
